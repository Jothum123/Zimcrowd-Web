Settings Tabs Force Show Script

Copy this script and paste it into your browser console while on the dashboard page:

// Force show all settings tabs
document.querySelectorAll('#settings-section .tab-content').forEach((tab, index) => {
    tab.style.display = 'block';
    tab.style.visibility = 'visible';
    tab.style.opacity = '1';
    tab.classList.add('active');
    console.log(`✅ Forced ${tab.id} to show`);
});
